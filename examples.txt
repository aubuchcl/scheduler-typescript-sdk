# 1 — smoke: minimal valid publish
R  node consume.js --max 1
P  node publish.js --topic smoke.test --payload '{}'

# 2 — real payload
R  node consume.js --max 1
P  node publish.js --topic orders.created --payload '{"id":"1234"}'

# 3 — topic filter, match
R  node consume.js --topics orders.created --max 1
P  node publish.js --topic orders.created --payload '{"id":"1234"}'

# 4 — topic filter, MISS: expect "sent":0, receiver keeps waiting (Ctrl-C it)
R  node consume.js --topics orders.created --max 1
P  node publish.js --topic billing.charged --payload '{"amount":100}'

# 5 — multiple topics
R  node consume.js --topics orders.created,orders.shipped --max 2
P  node publish.js --topic orders.created --payload '{"id":"1"}'
P  node publish.js --topic orders.shipped --payload '{"id":"1"}'

# 6 — named channel, matched
R  node consume.js --channel jobs --max 1
P  node publish.js --topic work.queued --payload '{"job":"a"}' --channel jobs

# 7 — channel MISMATCH: default publish, jobs receiver. Expect "sent":0 (Ctrl-C R)
R  node consume.js --channel jobs --max 1
P  node publish.js --topic work.queued --payload '{"job":"a"}'

# 8 — channel + topic together
R  node consume.js --topics work.queued --channel jobs --max 1
P  node publish.js --topic work.queued --payload '{"job":"b"}' --channel jobs

# 9 — annotations, pretty printed
R  node consume.js --max 1 --pretty
P  node publish.js --topic audit.event --payload '{"who":"chris"}' --annotation source=cli --annotation env=dev

# 10 — DURABILITY: run P FIRST, then R. Should replay the missed message.
P  node publish.js --topic orders.created --payload '{"id":"late"}' --ttl 120
R  node consume.js --topics orders.created --max 1

# 11 — no TTL: run P FIRST, then R. Should NOT arrive (Ctrl-C R)
P  node publish.js --topic orders.created --payload '{"id":"lost"}'
R  node consume.js --topics orders.created --max 1

# 12 — TTL EXPIRY: P first, wait 10s, then R. Should NOT arrive (Ctrl-C R)
P  node publish.js --topic ttl.short --payload '{"x":1}' --ttl 5
R  sleep 10; node consume.js --topics ttl.short --max 1

# 13 — non-JSON payload as a string
R  node consume.js --max 1
P  node publish.js --topic log.line --payload 'plain text, not json' --raw

# 14 — payload piped on stdin
R  node consume.js --max 1
P  echo '{"from":"stdin"}' | node publish.js --topic pipe.test

# 15 — nested + array payload
R  node consume.js --max 1 --pretty
P  node publish.js --topic batch.created --payload '{"ids":[1,2,3],"meta":{"src":"cli"}}'

# 16 — unicode round trip
R  node consume.js --max 1
P  node publish.js --topic i18n.test --payload '{"msg":"héllo wörld 日本語 🚀"}'

# 17 — fan-out: TWO receiver shells, then publish. Expect "sent":2
R  node consume.js --topics fanout.test --max 1        # shell A
R  node consume.js --topics fanout.test --max 1        # shell B
P  node publish.js --topic fanout.test --payload '{"n":1}'

# 18 — burst of 5
R  node consume.js --topics burst.test --max 5
P  for i in 1 2 3 4 5; do node publish.js --topic burst.test --payload "{\"n\":$i}"; done

# 19 — env vars instead of flags
R  TOPICS=env.test node consume.js --max 1
P  TOPIC=env.test PAYLOAD='{"v":1}' node publish.js

# 20 — bad key: both fail fast with 403, no retry loop
R  ACCESS_TOKEN=not-a-real-key node consume.js --max 1
P  ACCESS_TOKEN=not-a-real-key node publish.js --topic auth.test --payload '{}'
